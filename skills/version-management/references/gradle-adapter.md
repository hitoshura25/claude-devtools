# Gradle Version Adapter

## Purpose

Converts semantic versions to Android-compatible versionCode and syncs with `version.properties`.

## Script: `scripts/gradle-version.sh`

```bash
#!/bin/bash
set -euo pipefail

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
source "$SCRIPT_DIR/version-manager.sh" 2>/dev/null || true

# Convert semver to Android versionCode
# "1.2.34" → 1002034
semver_to_version_code() {
    local version="$1"
    IFS='.' read -r major minor patch <<< "$version"
    echo $((major * 1000000 + minor * 1000 + patch))
}

# Update version.properties file
update_version_properties() {
    local version="$1"
    local version_code=$(semver_to_version_code "$version")
    
    cat > version.properties << EOF
# Auto-generated by version scripts - do not edit manually
VERSION_NAME=$version
VERSION_CODE=$version_code
EOF
    
    echo "Updated version.properties:"
    echo "  VERSION_NAME=$version"
    echo "  VERSION_CODE=$version_code"
}

# Main
case "${1:-}" in
    generate)
        version=$(generate_version "${2:-patch}")
        echo "$version"
        if [[ -n "${GITHUB_OUTPUT:-}" ]]; then
            echo "version=$version" >> "$GITHUB_OUTPUT"
            echo "version-code=$(semver_to_version_code "$version")" >> "$GITHUB_OUTPUT"
        fi
        ;;
    update)
        version="${2:-$(get_latest_version)}"
        update_version_properties "$version"
        ;;
    version-code)
        version="${2:-$(get_latest_version)}"
        semver_to_version_code "$version"
        ;;
    *)
        echo "Usage: $0 {generate|update|version-code} [version|patch|minor|major]"
        ;;
esac
```

## version.properties

```properties
# Auto-generated by version scripts - do not edit manually
VERSION_NAME=1.2.3
VERSION_CODE=1002003
```

## build.gradle.kts Integration

```kotlin
import java.util.Properties

// Read version from version.properties
val versionProps = Properties().apply {
    val versionFile = rootProject.file("version.properties")
    if (versionFile.exists()) {
        versionFile.inputStream().use { load(it) }
    }
}

android {
    defaultConfig {
        versionName = versionProps.getProperty("VERSION_NAME", "0.0.1-dev")
        versionCode = versionProps.getProperty("VERSION_CODE", "1")?.toIntOrNull() ?: 1
    }
}
```

## Version Code Formula

```
versionCode = major * 1,000,000 + minor * 1,000 + patch

Examples:
1.0.0   → 1,000,000
1.2.3   → 1,002,003
2.15.42 → 2,015,042
```

This allows:
- 999 major versions
- 999 minor versions per major
- 999 patches per minor

## Usage

```bash
# Get current version
./scripts/gradle-version.sh version-code

# Generate and update
./scripts/gradle-version.sh generate patch
./scripts/gradle-version.sh update

# Update to specific version
./scripts/gradle-version.sh update 1.2.3
```

## GitHub Actions Integration

```yaml
- name: Generate version
  id: version
  run: |
    VERSION=$(./scripts/gradle-version.sh generate patch)
    ./scripts/gradle-version.sh update "$VERSION"
    echo "version=$VERSION" >> $GITHUB_OUTPUT

- name: Build release
  run: ./gradlew bundleRelease

- name: Tag release
  run: |
    git config user.name github-actions
    git config user.email github-actions@github.com
    git add version.properties
    git commit -m "chore: bump version to ${{ steps.version.outputs.version }}"
    git tag "v${{ steps.version.outputs.version }}"
    git push origin main
    git push origin "v${{ steps.version.outputs.version }}"
```

## Initial Setup

1. Create scripts directory:
   ```bash
   mkdir -p scripts
   ```

2. Create `scripts/version-manager.sh` (core script)

3. Create `scripts/gradle-version.sh` (this adapter)

4. Make executable:
   ```bash
   chmod +x scripts/*.sh
   ```

5. Create initial version.properties:
   ```bash
   ./scripts/gradle-version.sh update 1.0.0
   ```

6. Update build.gradle.kts to read from version.properties

7. Commit:
   ```bash
   git add scripts/ version.properties app/build.gradle.kts
   git commit -m "feat: add version management"
   git tag v1.0.0
   ```
